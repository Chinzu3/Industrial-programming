#include <iostream>
using namespace std;

int n;
int a[20];     
int cur[20];   

void gen(int i, int k) {
    if (i == n) {
        cout << "{ ";
        for (int j = 0; j < k; j++)
            cout << cur[j] << " ";
        cout << "}" << endl;
        return;
    }

    gen(i + 1, k);

    cur[k] = a[i];
    gen(i + 1, k + 1);
}

int main() {
    cin >> n;
    for (int i = 0; i < n; i++)
        cin >> a[i];

    gen(0, 0);

    return 0;
}

#include <iostream>
#include <fstream>
#include <string>
using namespace std;


int main() {
    string title[1000];    
    string author[1000];   
    int n = 0;

    ifstream in("books.txt");
    string line;
    while (getline(in, line)) {
        int pos = line.find(';');          
        title[n] = line.substr(0, pos);
        author[n] = line.substr(pos + 1);
        n++;
    }
    in.close();

    for (int i = 0; i < n - 1; i++)
        for (int j = 0; j < n - 1 - i; j++)
            if (author[j] > author[j + 1]) {
                swap(author[j], author[j + 1]);
                swap(title[j], title[j + 1]);
            }

    ofstream out("sorted_books.txt");
    for (int i = 0; i < n; i++)
        out << title[i] << ";" << author[i] << endl;
    out.close();

    return 0;
}
