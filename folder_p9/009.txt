#include <iostream>
#include <fstream>
#include <string>
#include <set>
using namespace std;

int main() {
    ifstream fin("input.txt");
    ofstream fout("output.txt");
    string text;
    getline(fin, text);
    string forbidden = "user agreement";

    for (char c: text) {
        char x = tolower(c);
        if (forbidden.find(x) == string::npos)
            fout << c;
    }

    fin.close();
    fout.close();
    return 0;
}

#include <iostream>
#include <fstream>
#include <string>
#include <bitset>
using namespace std;

int main() {
    ofstream fout("output.txt");
    int f = 0;
    ifstream fin("input.txt");
    string text, x = "", nums = "0123456789"; getline(fin, text);
    fin.close();

    for (char i: text){
        if (nums.find(i) != string::npos){
            f = 1;
            x += i;
        }else
            if (f){
                if (nums.find(i) != string::npos)
                    x += i;
                else{
                    fout << stoull(bitset<64>(stoi(x)).to_string()) << i;
                    f = 0; x = "";
                }
            }else
                fout << i;
    }
    if (x != "")
        fout << stoull(bitset<64>(stoi(x)).to_string());

    fout.close();
    return 0;
}
