#include <iostream>
#include <cmath>
using namespace std;

class math{
    public:
    static int next_ch(int x) {
            return x + (2 - x % 2);
    }
    static int des(int x){
            return ((x % 100) - (x % 10)) / 10;
    }
    static double sq(double x, double y){
            return sqrt(x * x + y * y);
    }
};

int main(){
    int a;
    double a1, b1;
    cin >> a;
    cout << math::next_ch(a) << endl;
    cout << math::des(a) << endl;
    cin >> a1 >> b1;
    cout << math::sq(a1, b1) << endl;
    return 0;
}
