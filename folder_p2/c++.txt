#include <iostream>
#include <vector>
using namespace std;


int main(){
    vector<int> a = {1, 4, 7, 10, 13};
    vector<int> b = {2, 3, 8, 12};
    int i = 0, j = 0;
    while ((i < a.size()) or (j < b.size())){
        if ((i < a.size()) and (j < b.size())){
            if (a[i] < b[j]){
                cout << a[i] << ' ';
                ++i;
            }else{
                cout << b[j] << ' ';
                ++j;
            }
        }
        else if (i < a.size()){
            cout << a[i] << ' ';
            ++i;
        }
        else{
            cout << b[j] << ' ';
            j++;
        }
    }
	return 0;
}

#include <iostream>
#include <algorithm>
#include <vector>
using namespace std;


int main(){
    int c = 0;
    vector<int> v = {0, 6, 3, 2, 10, 5, 1, 7, 9, 4};
    sort(v.begin(), v.end());
    int n = v.size();
    for(int i = 0; i <= n; ++i){
        if (c != v[i]){
            cout << c;
            break;
        }
        ++c;
    }
	return 0;
}
