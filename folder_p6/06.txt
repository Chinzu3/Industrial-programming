#include <iostream>
#include <vector>
using namespace std;

int main() {
    int n;
    cin >> n;
    vector<vector<int>> x(n, vector<int>(n, 0));
    for (int i = 0; i < n; ++i)
        for (int j = 0; j < n; ++j){
            if (j < (n - i - 1))
                x[i][j] = 0;
            else if (j == (n - i - 1))
                x[i][j] = 1;
            else if (j > (n - i - 1))
                x[i][j] = 2;
                
        }
    ////////////////
    for (int i = 0; i < n; ++i){
        for (int j = 0; j < n; ++j)
            cout << x[i][j] << ' ';
        cout << endl;
    }
    return 0;
}

#include <iostream>

int main() {
    int n;
    cin >> n;
    vector<vector<int>> x(n, vector<int>(n, 0));
    for (int i = 0; i < n; ++i)
        for (int j = 0; j < n; ++j){
            if (i + j >= n - 1)
                x[i][j] += 1;
            if (i + j >= n)
                x[i][j] += 1;
        }
    ////////////////
    for (int i = 0; i < n; ++i){
        for (int j = 0; j < n; ++j)
            cout << x[i][j] << ' ';
        cout << endl;
    }
    return 0;
}

#include <iostream>
#include <vector>
#include <string>
using namespace std;

int main() {
    int n, m, w, x, y;
    cin >> n >> m >> w;
    vector<vector<string>> v(n, vector<string>(m, "0"));
    for (int k = 0; k < w; ++k){
        cin >> x >> y;
        v[x - 1][y - 1] = "*";
        for (int i = (x - 2) * (x - 1 != 0) + (x - 1 == 0) * (x - 1); i <= x * (x - 1 != n - 1) + (x - 1 == n - 1) * (x - 1); ++i){
            for (int j = (y - 2) * (y - 1 != 0) + (y - 1 == 0) * (y - 1); j <= y * (y - 1 != m - 1) + (y - 1 == m - 1) * (y - 1); ++j)
                if (v[i][j] != "*")
                    v[i][j] = to_string(stoi(v[i][j]) + 1);
        }
    }
    /////////////
    for (int i = 0; i < n; ++i){
        for (int j = 0; j < m; ++j)
            cout << v[i][j] << ' ';
        cout << endl;
    }
    return 0;
}

#include <iostream>
#include <vector>
#include <string>
using namespace std;

int main() {
    int n, c; string x;
    cin >> x >> n;
    string y = x;
    /////////////
    for (int j = 1; j <= n - 1; ++j){
        y = "";
        for (int i = 0; i < x.size(); ++i){
            if (x[i - 1] == x[i])
                continue;
            c = 1;
            for (int k = i + 1; k < x.size(); ++k)
                if (x[k] == x[i])
                    c += 1;
                else
                    break;
            y += to_string(c) + x[i];
        }
        x = y;
    }
    /////////////
    cout << y;
    return 0;
}
